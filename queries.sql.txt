-- 1. Abonentlərin son 3 ayda istifadə etdiyi xidmətlərin adlarını və istifadə sayını göstər:
Ekrana abonentin adı, soyadı, xidmətin adı və istifadə sayı çıxsın.

select s.name,s.surname,sr.service_name,count(c.service_id) as say
from subscribers s join calls c
on s.subscribers_id =c.caller_id 
join services sr
on c.service_id=sr.service_id
where c.call_date > add_months(sysdate, -3)
group by s.name,s.surname,sr.service_name;
      
--2.Hər abonent üçün son ödəniş tarixini və məbləğini göstər: 
--    Ekrana Ad, soyad, son ödəniş tarixi, məbləği və ödənişin üsulu haqqında informasiyalar çıxsın.
            
select s.name,s.surname,pp.payment_date,pp.amount,pt.name
from subscribers s inner join payments pp
on s.subscribers_id=pp.subscribers_id
inner join payment_method_type pt
on pp.payment_type_id=pt.payment_type_id
where pp.payment_date=(select max(p.payment_date) from payments p where p.subscribers_id=s.subscribers_id);
             
-- 3.Hər abonent üçün edilən zənglərin sayını və ümumi zəng müddətini göstər:
--    Ekrana ad, soyad, abonent üçün edilən zənglərin sayını, ümumi zəng müddətini, zəngin tipi haqqında informasiyalar çıxsın.

select s.name ad,s.surname soyad,ct.name zeng_tipi,sum(call_duration) as umumi_zeng_muddeti,count(c.receiver_phone) as zeng_sayi
from subscribers s inner join calls c 
on s.subscribers_id=c.caller_id
join call_type ct 
on c.call_type_id=ct.call_type_id
group by s.name, s.surname,ct.name;
 
--4.Hər abonent üçün tarifə görə aylıq ödədikləri məbləği və zənglərin sayını göstər:
--    Ekrana ad, soyad,tarifə görə aylıq ödədikləri məbləğ və zənglərin sayı haqqında informasiyalar çıxsın.

select s.name,s.surname,ti.monthly_subscription,count(c.call_id) as zeng_sayi
from subscribers s inner join calls c
on s.subscribers_id=c.caller_id 
join services sr
on c.service_id=sr.service_id
join tariff_informations ti
on sr.tariff_id=ti.tariff_id
group by s.name,s.surname,ti.monthly_subscription; 
         
--5.Zənglərin növünə görə abonentlərin sayını və ümumi zəng müddətini göstər:
select ct.name, count(distinct(s.subscribers_id)) abonent_sayi, sum(c.call_duration) umumi_zeng_muddeti
from call_type ct inner join calls c
on ct.call_type_id = c.call_type_id 
join subscribers s
on c.caller_id=s.subscribers_id 
group by ct.name;
            
--6. Aktiv olmayan, lakin son 6 ayda ödəniş etmiş abonentləri tap.
Ekrana ad, soyad, status və ödəniş tarixi çıxsın.

select concat(s.name,' ',  s.surname) as full_name,s.status,to_char(p.payment_date,'dd.mm.yyyy')
from subscribers s inner join payments p
on s.subscribers_id=p.subscribers_id
where s.status='PASSIVE' AND MONTHS_BETWEEN(SYSDATE,P.PAYMENT_DATE)<=6;
              
--7.Son 6 ayda edilən ödənişlərin məbləğini və xidmətlərin sayını abonentlər üzrə göstər.
--    Ekrana abonentin adı, abonentin soyadı, xidmətlərin sayı və 6 ayda edilən ödənişlərin məbləği haqqında informasiyalar çıxsın.
   
select s.name, s.surname, 
(select sum(p.amount) from payments p where p.subscribers_id = s.subscribers_id 
and p.payment_date > add_months(sysdate, -6) ) as odenis_meblegi, 
(select count(c.service_id) from calls c where c.caller_id  = s.subscribers_id
and c.call_date > add_months(sysdate, -6 )) as xidmet_sayi
from  subscribers s ;
     
--8.Abonentlərdən ən çox şikayət edən 10 nəfərin adını və şikayət sayını göstər:
Ekrana ad, soyad və ümumi şikayət sayı çıxsın.

Həll 1 – join 
select s.name,s.surname,count(csr.description ) as say
from subscribers s inner join complaints_and_support_requests csr
on s.subscribers_id=csr.subscribers_id 
where lower(csr.description) like lower('%issue%')
group by s.name,s.surname 
order by say desc
fetch next 10 rows only;

Həll 2 - subquery
select s.name,s.surname,
(select count(csr.subscribers_id) from complaints_and_support_requests csr where s.subscribers_id=csr.subscribers_id
and lower(DESCRIPTION) like '%issue%') as sikayet_sayi
from SUBSCRIBERS s
order by sikayet_sayi desc
fetch next 10 rows only;


Həll 3 – with 
with say as(select subscribers_id,count(subscribers_id) as sikayet_sayi from complaints_and_support_requests
where lower(DESCRIPTION) like '%issue%'
group by subscribers_id) select name, surname, sikayet_sayi from SUBSCRIBERS s join say t
on s.SUBSCRIBERS_ID=t.subscribers_id
order by sikayet_sayi desc
fetch next 10 rows only;
 
--9. Hər abonent üçün son 12 ayda göndərilən SMS-lərin sayını və SMS məzmununu göstər:

select s.name,s.surname,si.sms_content as sms_mezmunu,count(*) as sms_sayi
from subscribers s inner join sms_informations si
on s.phone_number=si.receiver_phone
where months_between(sysdate,si.insert_date)<=12
group by s.name,s.surname,si.sms_content;
     
--10.Hər abonent üçün edilən zənglərin ümumi müddətini və ödəniş məlumatlarını göstər:    
--   Ekrana abonentin adı, abonentin soyadı, zənglərin ümumi müddətini ödəniş məlumatları haqqında informasiyalar çıxsın. 

select s.name,s.surname,
(select nvl(sum(c.call_duration),0) from calls c where s.phone_number=c.receiver_phone
and c.call_type='Receive') as zeng_muddeti,
(select nvl(sum(p.amount),0) from payments p where s.subscribers_id=p.subscribers_id) as umumi_odenis
from subscribers s;
      
--11. Hər tarif üzrə abonentlərin aylıq ödədikləri məbləğin orta dəyərini və zənglərin sayını göstər:
--   Ekrana tarif adı, abonentlərin aylıq ödədikləri məbləğin orta dəyəri və zənglərin sayı haqqında informasiyalar çıxsın. 

select ti.tariff_name,avg(ti.monthly_subscription) as ayliq_orta_deyer, count(c.call_id) as zeng_sayi
from calls c inner join services s
on c.service_id=s.service_id
join tariff_informations ti
on s.tariff_id=ti.tariff_id
group by ti.tariff_name;

--12. Hər abonent üçün son 12 ayda göndərilən SMS-lərin məzmununu və göndərilən SMS növlərini göstər:
--   Ekrana abonentin adı, soyadı, son 12 ayda göndərilən SMS-lərin məzmununu və göndərilən SMS növləri haqqında informasiyalar çıxsın. 

select s.name,s.surname,si.sms_content as sms_mezmunu,st.name as sms_novu
from subscribers s inner join sms_informations si
on s.phone_number=si.receiver_phone
join sms_types st
on si.sms_type_id=st.sms_type_id
where months_between(sysdate,si.insert_date)<=12; 
      
-- 13. Hər abonent üçün son 6 ayda edilən ödənişlərin məbləğini və xidmətlərin sayını göstər:
--   Ekrana abonentin adı, soyadı, son 6 ayda edilən ödənişlərin məbləğini və xidmətlərin sayı haqqında informasiyalar çıxsın.

Həll 1 - join
select s.name,s.surname,sum(amount),count(p.subscribers_id)
from subscribers s inner join payments p
on s.subscribers_id=p.subscribers_id
where months_between(sysdate,payment_date)<=6
group by s.name,s.surname;

Həll 2 - subquery
select s.name, s.surname,
       (select nvl(sum(p.amount), 0) from payments p 
        where p.subscribers_id = s.subscribers_id 
          and months_between(sysdate, p.payment_date) <= 6) AS umumi_odenis,
       (select count(*) from payments p 
        where p.subscribers_id = s.subscribers_id 
          and months_between(sysdate, p.payment_date) <= 6) AS xidmet_sayi
from subscribers s;
       
-- 14. Hər abonent üçün ödənişlər və şikayətlərin məbləğini göstər:
--   Ekrana abonentin adı, soyadı, hər abonent üçün ödənişlər və şikayətlərin sayı haqqında informasiyalar çıxsın. 

select s.name,s.surname,
(select count(*) from complaints_and_support_requests csr where s.subscribers_id =csr.subscribers_id) as sikayet_sayi,
(select nvl(sum(p.amount),0) from payments p where p.subscribers_id =s.subscribers_id) as odenisler
from subscribers s;
      
-- 15. Abonentlərin yaş qrupuna görə və cinsiyyətə görə bölgüsünü göstər:
--    Yaş qrupları aşağıdakı şəkildədi
--    18 və aşağı
--    19-30
--    31-50
--    51 və yuxarı
--    Ekrana yaş aralığı,cins və say haqqında informasiyalar çıxsın.  

select gender,count(*) as say,
case when trunc((sysdate-birth_date)/365)<=18 then '18 ve asagi'
    when trunc((sysdate-birth_date)/365) between 19 and 30 then '19-30'
    when trunc((sysdate-birth_date)/365) between 31 and 50 then '31-50'
    else '51 ve yuxari'
    end as yas_qruplari
from subscribers
group by gender,case when trunc((sysdate-birth_date)/365)<=18 then '18 ve asagi'
    when trunc((sysdate-birth_date)/365) between 19 and 30 then '19-30'
    when trunc((sysdate-birth_date)/365) between 31 and 50 then '31-50'
    else '51 ve yuxari'
    end;
